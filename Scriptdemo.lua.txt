automacro autoBlock {
    ConfigKey NotWhileSitting 1
    exclusive 1
    run-once 1
    
    call {
        do conf attackAuto 2
        do conf attackAuto_party 1
        do conf attackAuto_only 1
        do conf attackAuto_followTarget 1
        do conf attackDistance 1.5
        do conf attackMaxDistance 2.5
        do conf attackMaxRouteDistance 15
        do conf attackMinPlayerDistance 3
        do conf attackMinPortalDistance 4
        do conf attackUseWeapon 1
        do conf autoMoveOnDeath 0
    }
}

automacro changeMap {
    location current
    configKey NotWhileSitting 1
    timeout 60
    
    call {
        $random = @random (1, 4)
        if ($random == 1) do move 50 50
        if ($random == 2) do move 100 100
        if ($random == 3) do move 150 150
        if ($random == 4) do move 200 200
        pause 2
    }
}

automacro knightSetup {
    JobID = 7    # Knight
    run-once 1
    
    call {
        do conf tankMode 1
        do conf tauntOnMob 1
        do conf useSelf_skill Bash {
            lvl 10
            whenStatusInactive EFST_POSTDELAY
        }
    }
}

automacro priestSetup {
    JobID = 6    # Priest
    run-once 1
    
    call {
        do conf useSelf_skill Heal {
            lvl 10
            hp < 80%
        }
        do conf useSelf_skill Blessing {
            lvl 10
            whenStatusInactive EFST_BLESSING
        }
    }
}

automacro wizardSetup {
    JobID = 9    # Wizard
